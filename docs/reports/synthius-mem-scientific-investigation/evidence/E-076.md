### 1. Context formatter pseudocode

```python
import json
from typing import Dict, List

def format_context(per_domain_facts: Dict[str, List[FactItem]],
                   planner_domain_order: List[str]) -> str:
    sections = []
    for domain in planner_domain_order:
        facts = per_domain_facts.get(domain, [])
        if not facts: continue
        pretty = json.dumps(
            [{"fact_id": f.fact_id, "fields": f.fields,
              "confidence": f.envelope["confidence"]}
             for f in facts],
            ensure_ascii=False, indent=0, separators=(",",":"))  # compact
        section = f"## {domain.title().replace('_',' ')}\n{pretty}\n"
        sections.append(section)
    return "\n".join(sections)
```
