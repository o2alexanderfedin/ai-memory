## Drafted Two-Stage Pipeline (pseudocode)

```python
# Stage 1: Deterministic candidate generation
NICKNAME_TABLE = {
    "mel":     ["melanie","melissa"],
    "bob":     ["robert","bobby"],
    "rob":     ["robert"],
    "caro":    ["caroline","carolina"],
    "beth":    ["elizabeth"],
    "liz":     ["elizabeth"],
    "mom":     [],            # role terms — empty deterministic target
    "dad":     [],
    "mother":  [],
    "father":  [],
}

def normalize(s: str) -> str:
    s = s.lower().strip()
    s = re.sub(r"[^\w\s]", "", s)       # strip punctuation (Dr., St., Jr.)
    s = re.sub(r"\b(dr|mr|mrs|ms|prof|sir|lord|madam)\b", "", s).strip()
    s = re.sub(r"\s+", " ", s)
    return s

def candidate_matches(mention: str, existing_persons: list[Person]) -> list[Person]:
    mn = normalize(mention)
    cands = []
    for p in existing_persons:
        all_forms = [normalize(n) for n in p.names + p.aliases]
        # Exact normalized match
        if mn in all_forms:
            cands.append((p, "exact_normalized"))
            continue
        # First-name token match
        if mn.split()[0] in {f.split()[0] for f in all_forms if f}:
            cands.append((p, "first_token"))
            continue
        # Nickname table expansion
        expansions = NICKNAME_TABLE.get(mn, [])
        if any(exp in " ".join(all_forms) for exp in expansions):
            cands.append((p, "nickname_table"))
            continue
        # Reverse: is existing a known nickname of mention?
        for f in all_forms:
            nicks = [k for k,v in NICKNAME_TABLE.items() if f.split()[0] in v]
            if mn in nicks:
                cands.append((p, "reverse_nickname"))
                break
    return cands

# Stage 2: LLM adjudication
def llm_adjudicate(mention: str, context: str, candidates: list[Person]) -> Person | None:
    prompt = f"""Given the mention "{mention}" in context:
---
{context}
---
And existing persons: {[(p.person_id, p.names, p.aliases, p.relationship) for p in candidates]}
Decide: is "{mention}" one of these persons, or a new person?
Respond JSON: {{"decision": "match" | "new", "person_id": "<id>" | null, "reasoning": "..."}}
"""
    return llm.generate(prompt, response_format="json")

def resolve(mention: str, context: str, registry: list[Person]) -> Person:
    cands = candidate_matches(mention, registry)
    if not cands:
        return Person.new(mention)         # Stage-1 miss → new person, no LLM call
    if len(cands) == 1 and cands[0][1] == "exact_normalized":
        return cands[0][0]                  # High-confidence deterministic match, no LLM
    verdict = llm_adjudicate(mention, context, [c[0] for c in cands])
    if verdict["decision"] == "match":
        p = registry[verdict["person_id"]]
        p.aliases.append(mention) if mention not in p.aliases + p.names else None
        return p
    return Person.new(mention)
```
