## Drafted signature

```python
AVG_TOKENS = {"biography":80,"experiences":140,"preferences":70,
              "social_circle":90,"work":110,"psychometrics":200}
CAP = 2000

def derive_k(domain: str, selected_domains: int = 1) -> int:
    budget = CAP // selected_domains
    return min(20, budget // AVG_TOKENS[domain])

def retrieve(persona_id: str, domain: str, field: str, value,
             op: str = "eq", limit: int | None = None) -> list[FactItem]:
    k = limit or derive_k(domain)
    # Rank by (recency DESC, confidence DESC); return top-k.
    return INDEX[domain].probe(persona_id, field, value, op)[:k]
```
