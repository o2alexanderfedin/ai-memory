### Canonical target schema (shared output contract)

```python
from dataclasses import dataclass
from typing import List, Optional, Literal
from datetime import datetime

@dataclass(frozen=True)
class Attachment:
    type: Literal["image","voice","video","file","link"]
    uri: str          # opaque blob-store URI; never inlined
    mime: str

@dataclass(frozen=True)
class Message:
    speaker: str               # normalized display name
    ts: str                    # ISO-8601 UTC with 'Z'
    text: str                  # plain-text body (no HTML)
    attachments: Optional[List[Attachment]] = None
    source_format: str = ""    # {"whatsapp","telegram","pdf","email"}
    message_id: str = ""       # ULID (assigned post-parse by H-063; stable hash pre-assignment)
```

### 1. WhatsApp `_chat.txt` line parser

```python
# Format: "[2024-03-15, 10:22:01] Alice: I just got back from UCL <Media omitted>"
WA_LINE = re.compile(r"^\[(?P<d>\d{4}-\d{2}-\d{2}),\s*(?P<t>\d{2}:\d{2}:\d{2})\]\s+(?P<who>[^:]+):\s(?P<body>.*)$")
def parse_whatsapp(stream: Iterable[str]) -> Iterable[Message]:
    buf = None
    for line in stream:
        m = WA_LINE.match(line)
        if m:
            if buf: yield buf
            attachments = []
            body = m["body"]
            if "<Media omitted>" in body:
                attachments = [Attachment(type="file", uri="placeholder://media", mime="application/octet-stream")]
                body = body.replace("<Media omitted>","").strip()
            ts = f"{m['d']}T{m['t']}Z"
            buf = Message(speaker=m["who"].strip(), ts=ts, text=body,
                          attachments=attachments or None, source_format="whatsapp")
        else:
            # continuation line for multi-line body
            if buf:
                buf = replace(buf, text=(buf.text + "\n" + line.rstrip()))
    if buf: yield buf
```

### 2. Telegram JSON-export parser

```python
def parse_telegram(path: str) -> Iterable[Message]:
    doc = json.load(open(path))
    for m in doc["messages"]:
        if m["type"] != "message": continue
        attachments = []
        if "photo" in m:         attachments.append(Attachment("image", m["photo"], "image/jpeg"))
        if "file" in m:          attachments.append(Attachment("file", m["file"], m.get("mime_type","application/octet-stream")))
        if "media_type" == "voice_message":
            attachments.append(Attachment("voice", m["file"], "audio/ogg"))
        text = m["text"] if isinstance(m["text"], str) else "".join(t if isinstance(t,str) else t.get("text","") for t in m["text"])
        yield Message(speaker=m["from"], ts=m["date"] + ("Z" if not m["date"].endswith("Z") else ""),
                      text=text, attachments=attachments or None, source_format="telegram")
```

### 3. PDF email-thread parser (thin — no LLM)

```python
# Uses pdfminer.six for text extraction; quoted-reply is detected by ">" prefix.
def parse_pdf_email(path: str) -> Iterable[Message]:
    pages = extract_text(path).split("\f")
    # header heuristic: "From: X\nDate: RFC 2822\nSubject: ..."
    for block in re.split(r"\n(?=From:\s)", "\n".join(pages)):
        hdrs, _, body = block.partition("\n\n")
        h = dict(re.findall(r"^([A-Za-z-]+):\s(.+)$", hdrs, re.M))
        speaker = h.get("From","unknown").strip()
        ts = rfc2822_to_iso8601(h["Date"]) if "Date" in h else ""
        # strip quoted-reply levels (">" or ">>")
        clean = "\n".join(ln for ln in body.splitlines() if not ln.lstrip().startswith(">"))
        yield Message(speaker=speaker, ts=ts, text=clean.strip(), source_format="pdf")
```

### 4. MIME email parser

```python
def parse_mime(path: str) -> Iterable[Message]:
    msg = email.message_from_file(open(path))
    speaker = msg["From"]
    ts = rfc2822_to_iso8601(msg["Date"])
    text_parts, attachments = [], []
    for part in msg.walk():
        ct = part.get_content_type()
        if ct.startswith("text/"):
            text_parts.append(part.get_payload(decode=True).decode(errors="replace"))
        elif part.get_content_disposition() == "attachment" or ct.startswith("image/"):
            cid = part.get("Content-ID","").strip("<>")
            attachments.append(Attachment(
                type="image" if ct.startswith("image/") else "file",
                uri=f"cid:{cid or uuid.uuid4()}", mime=ct))
    yield Message(speaker=speaker, ts=ts, text="\n".join(text_parts),
                  attachments=attachments or None, source_format="email")
```

### Idempotency hash (re-ingest dedup)

```python
def msg_hash(m: Message) -> str:
    return hashlib.sha256(f"{m.speaker}|{m.ts}|{m.text}".encode()).hexdigest()
```
