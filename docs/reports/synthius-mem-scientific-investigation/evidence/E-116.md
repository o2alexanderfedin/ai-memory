### Artifact 7: Storage schema DDL (per T-019)

```sql
CREATE TABLE facts (
  fact_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  persona_id UUID NOT NULL,
  domain TEXT NOT NULL CHECK (domain IN (
    'biography','experiences','preferences','social_circle','work','psychometrics'
  )),
  payload JSONB NOT NULL,
  confidence FLOAT CHECK (confidence BETWEEN 0 AND 1),
  provenance JSONB NOT NULL,
  schema_version INT NOT NULL DEFAULT 1,
  created_at TIMESTAMPTZ NOT NULL DEFAULT NOW(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT NOW()
);

CREATE INDEX idx_facts_persona_domain ON facts (persona_id, domain);
CREATE INDEX idx_facts_jsonb_gin ON facts USING GIN (payload jsonb_path_ops);
CREATE INDEX idx_facts_biography_category ON facts 
  ((payload->>'category')) 
  WHERE domain = 'biography';

CREATE TABLE fact_diffs (
  diff_id UUID PRIMARY KEY,
  persona_id UUID NOT NULL,
  prev_state JSONB,
  new_state JSONB,
  commit_ts TIMESTAMPTZ NOT NULL,
  operation TEXT CHECK (operation IN ('insert','update','delete'))
);

CREATE INDEX idx_diffs_persona ON fact_diffs (persona_id, commit_ts DESC);
```

### Artifact 8: CI gate (GitHub Actions YAML)

```yaml
# .github/workflows/eval-gate.yml
name: LoCoMo Eval Gate
on: [pull_request]
jobs:
  eval:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: docker/setup-buildx-action@v3
      - run: docker build -t synthius-mem .
      - run: docker run --rm synthius-mem eval --dataset locomo_v1 --output /tmp/results.json
      - name: Parse accuracy
        run: |
          ACC=$(jq -r '.weighted_accuracy' /tmp/results.json)
          REGRESS=$(echo "$ACC < 93.37" | bc -l)
          if [ "$REGRESS" = "1" ]; then
            echo "Accuracy regression: $ACC < baseline 93.37 (-1pp margin)"
            exit 1
          fi
```
