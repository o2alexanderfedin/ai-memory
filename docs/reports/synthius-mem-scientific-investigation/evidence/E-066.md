### 1. Three-stage consolidation pipeline

```python
from typing import List, Tuple, Literal
import hashlib, json

# ==============  Stage 1: Deterministic dedup  ==============
def dedup_key(fact: dict, domain: str) -> str:
    """Composite typed-field key per domain. Exact match on this key = duplicate."""
    if domain == "biography":
        f = fact["fields"]
        payload = f"{f.get('category')}|{f.get('degree','')}|{f.get('field','')}|{f.get('institution','')}|{f.get('year','')}"
    elif domain == "work":
        f = fact["fields"]
        payload = f"{f.get('employer','')}|{f.get('role','')}|{f.get('start','')}|{f.get('project','')}"
    elif domain == "social_circle":
        # person-level identity: canonical name after case-fold + alias-resolve
        payload = canonicalize_person_name(fact["fields"].get("names",[""])[0])
    elif domain == "experiences":
        f = fact["fields"]
        payload = f"{f.get('summary','')[:120]}|{f.get('start','')}"
    elif domain == "preferences":
        f = fact["fields"]
        payload = f"{f.get('target','')}|{f.get('polarity','')}"
    elif domain == "psychometrics":
        f = fact["fields"]
        payload = f"{f.get('framework','')}|{f.get('trait','')}"
    return hashlib.sha256(payload.lower().encode()).hexdigest()

def stage1_dedup(facts: List[dict], domain: str) -> List[dict]:
    seen = {}
    for f in facts:
        k = dedup_key(f, domain)
        if k in seen:
            # keep the one with higher confidence; older seq wins tie-break (stable)
            if f["envelope"]["confidence"] > seen[k]["envelope"]["confidence"]:
                seen[k] = f
        else:
            seen[k] = f
    return list(seen.values())

# ==============  Stage 2: Non-conflict merge  ==============
def stage2_merge(existing: dict, incoming: dict) -> dict:
    """Merge two facts that share dedup_key but have disjoint non-empty fields."""
    merged_fields = {**existing["fields"], **incoming["fields"]}
    # Detect conflicts: same key, different non-null values
    for k in existing["fields"]:
        if k in incoming["fields"] and incoming["fields"][k] != existing["fields"][k]:
            return None   # signal conflict — escalate to Stage 3
    return {
        "fact_id": existing["fact_id"],   # keep existing id (stable)
        "fields": merged_fields,
        "envelope": {
            **existing["envelope"],
            "confidence": max(existing["envelope"]["confidence"],
                              incoming["envelope"]["confidence"]),
            "provenance": existing["envelope"]["provenance"] + incoming["envelope"]["provenance"],  # append
        }
    }

# ==============  Stage 3: LLM conflict escalation  ==============
RESOLVE_CONFLICT_SCHEMA = {
  "type":"object","additionalProperties":False,
  "properties":{
    "decision":{"type":"string","enum":["merge","supersede","keep_both","reject"]},
    "merged":{"type":"object"},          # populated only when decision=merge|supersede
    "rationale":{"type":"string"}
  },
  "required":["decision","rationale"]
}

def stage3_resolve(existing: dict, incoming: dict, context_snippet: str) -> dict:
    result = openai.chat.completions.parse(
      model="gpt-4.1-mini",
      response_format=RESOLVE_CONFLICT_SCHEMA,
      messages=[
        {"role":"system","content":"Resolve the conflict between two versions of the same fact."},
        {"role":"user","content":json.dumps({
          "existing": existing, "incoming": incoming, "context": context_snippet
        })}
      ]
    ).choices[0].message.parsed
    return apply_resolution(existing, incoming, result)

# ==============  Orchestrator  ==============
def consolidate(existing: List[dict], incoming: List[dict], domain: str) -> List[dict]:
    combined = stage1_dedup(existing + incoming, domain)
    by_key = {}
    for f in combined:
        k = dedup_key(f, domain)
        if k in by_key:
            merged = stage2_merge(by_key[k], f)
            if merged is None:
                merged = stage3_resolve(by_key[k], f, context_snippet_for(f))
            by_key[k] = merged
        else:
            by_key[k] = f
    return list(by_key.values())
```
